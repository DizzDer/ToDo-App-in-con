cmake_minimum_required(VERSION 3.16)
project(todo_manager VERSION 1.0.0 LANGUAGES CXX)
add_library(todo_core todo.cpp TodoExtras.cpp)
target_compile_features(todo_core PUBLIC cxx_std_17)
target_include_directories(todo_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(todo main.cpp)
target_link_libraries(todo PRIVATE todo_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(todo_tests tests/todo_tests.cpp)
  target_link_libraries(todo_tests PRIVATE todo_core)
  add_test(NAME todo_tests COMMAND todo_tests)
endif()
